  <script>
    // 初始化時從瀏覽器的 localStorage 讀取紀錄
    let poopLogs = JSON.parse(localStorage.getItem('poopLogs')) || [];

    function renderLogs() {
      const list = document.getElementById('logList');
      list.innerHTML = '';
      
      if (poopLogs.length === 0) {
        list.innerHTML = '<li style="color: #aaa; justify-content: center;">目前還沒有紀錄喔！</li>';
        return;
      }
      
      // 反轉陣列，讓最新的一筆紀錄顯示在最上面
      const reversedLogs = [...poopLogs].reverse();
      
      reversedLogs.forEach(log => {
        const li = document.createElement('li');
        li.innerHTML = `<span class="dog-name">${log.name}</span> <span class="time-stamp">${log.time}</span>`;
        list.appendChild(li);
      });
    }

    function logPoop(dogName) {
      const now = new Date();
      // 格式化時間：MM/DD HH:mm (例如：12/24 08:30)
      const timeString = `${now.getMonth() + 1}/${now.getDate()} ${now.getHours().toString().padStart(2, '0')}:${now.getMinutes().toString().padStart(2, '0')}`;
      
      poopLogs.push({ name: dogName, time: timeString });
      localStorage.setItem('poopLogs', JSON.stringify(poopLogs));
      
      renderLogs();
    }

    function clearLogs() {
      if (confirm('確定要清除所有的便便紀錄嗎？')) {
        poopLogs = [];
        localStorage.removeItem('poopLogs');
        renderLogs();
      }
    }

    // 網頁一載入時先顯示現有紀錄
    renderLogs();
  </script>
